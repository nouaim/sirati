%-------------------------------------------------------------------------------
%  Layout

\newcommand{\cvsection}[1]{%
  \par\vspace{7.5mm}%
  {\noindent\fontsize{14.5pt}{1.2em}\bfseries\color{awesome}#1\phantomsection}%
  \hspace{0.6em}{\color{sectiondivider}\leaders\hrule height 0.8pt\hfill\kern0pt}\par%
  \vspace{2.8mm}%
}

\newenvironment{cvparagraph}{%
  \vspace{-1mm}%
  \fontsize{10.2pt}{1.45em}\color{darktext}%
}{%
  \par\vspace{1.6mm}%
}

\newenvironment{cventries}{%
  \vspace{0.4mm}%
}{%
  \vspace{1.4mm}%
}

% #1 position, #2 organisation, #3 location, #4 dates, #5 details
\newcommand*{\cventry}[5]{%
  \vspace{3.6mm}%
  \noindent{\bfseries\fontsize{11pt}{1.2em}\color{darktext} #2}%
  \hfill{\fontsize{9.4pt}{1.1em}\textcolor{awesome}{#3}}\\%
  \noindent{\fontsize{9.6pt}{1.1em}\color{graytext} #1}%
  \hfill{\fontsize{9pt}{1.1em}\color{graytext} #4}%
  \par
  \if\relax\detokenize{#5}\relax\else
    \vspace{0.8mm}%
    #5%
  \fi
  \vspace{0.6mm}%
}

\newenvironment{cvitems}{%
  \vspace{-0.8mm}%
  \begin{itemize}[leftmargin=1.5em, itemsep=0.28em, topsep=0.1em, parsep=0pt]
  \fontsize{9.7pt}{1.35em}\color{darktext}%
}{%
  \end{itemize}%
  \vspace{0.6mm}%
}

\newenvironment{cvskills}{%
  \vspace{0.4mm}%
}{%
  \vspace{0.6mm}%
}

% Skill line: the ambient direction is already RTL, so the line is right-ragged
% with no skip trickery.  The paragraph must close *outside* the group: a group
% that spans \par leaves bidi with an unbalanced \endL and it warns about it.
\newcommand*{\cvskill}[2]{%
  \noindent{\fontsize{10.2pt}{1.4em}\selectfont
    \textbf{\color{darktext}#1:~}\color{darktext}#2}%
  \par\vspace{2.6mm}%
}
