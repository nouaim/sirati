% The only place in this repository where the font families are named.  Both
% documents input this file, so changing a font is one line here rather than two
% near-identical preambles.
%
% The families are referenced by name and no font files are bundled, so they have
% to be installed on the system: apt provides Roboto through fonts-roboto, while
% the Arabic family here has no distribution package and is downloaded by
% install.sh and by the Dockerfile.
%
% If you change the Arabic family, expect to revisit its Scale= too: 0.96
% compensates for Tajawal's metrics, and another face will sit differently against
% the Latin text and the 10pt grid.
%
% Any change must leave both documents on one page, with no missing glyphs and no
% bidi warnings — tests/check-arabic-examples.sh measures all three, and the CI
% and Docker jobs must be given the new family as well.
%
% Packaged Arabic faces worth testing: Amiri (fonts-hosny-amiri) and Noto Naskh
% Arabic (fonts-noto-core).  Neither has been built here yet.
%
% The icon set and its style are the exception to "families are named here and
% nowhere else": they are chosen before the document is read, by
%
%   make cv-ar ICONS=material STYLE=outlined
%
% or by the same two \def lines on the XeLaTeX command line.  Font Awesome arrives
% with its CTAN package, so the default set names no family at all; Material Icons
% is Google's set, installed by install.sh, and carries one family per style —
% only the style in use is defined, so a machine needs the family it builds rather
% than all five.  icons.tex holds the glyphs each set draws with.
%-------------------------------------------------------------------------------
\defaultfontfeatures{Ligatures=TeX}
\newfontfamily\arabicfont[Script=Arabic, Scale=0.96]{Tajawal}
\newfontfamily\arabicfontsf[Script=Arabic, Scale=0.96]{Tajawal}
\newfontfamily\englishfont[Scale=0.92]{Roboto}

\providecommand*{\cvIconSet}{fa}
\providecommand*{\cvIconStyle}{filled}
\def\cvIconSetMaterial{material}
\def\cvIconStyleFilled{filled}
\def\cvIconStyleOutlined{outlined}
\def\cvIconStyleRound{round}
\def\cvIconStyleSharp{sharp}
\def\cvIconStyleTwoTone{twotone}
% Font Awesome's names for the first two, accepted as aliases so either vocabulary
% works on the command line.
\def\cvIconStyleSolid{solid}
\def\cvIconStyleRegular{regular}
\ifx\cvIconStyle\cvIconStyleSolid\def\cvIconStyle{filled}\fi
\ifx\cvIconStyle\cvIconStyleRegular\def\cvIconStyle{outlined}\fi
\ifx\cvIconSet\cvIconSetMaterial
  \def\cvIconFace{Material Icons}
  \ifx\cvIconStyle\cvIconStyleOutlined\def\cvIconFace{Material Icons Outlined}\fi
  \ifx\cvIconStyle\cvIconStyleRound\def\cvIconFace{Material Icons Round}\fi
  \ifx\cvIconStyle\cvIconStyleSharp\def\cvIconFace{Material Icons Sharp}\fi
  \ifx\cvIconStyle\cvIconStyleTwoTone\def\cvIconFace{Material Icons Two Tone}\fi
  % Material Icons has no space glyph, and fontspec typesets one while setting the
  % family up: that alone fills the log with "Missing character: There is no
  % (U+0020)".  Silence the parameter across the definition only, so a genuinely
  % missing glyph anywhere else is still reported.  Scale=1.10 and the shift in
  % icons.tex were measured against the Font Awesome row at this size.
  \tracinglostchars=0
  \newfontfamily\iconfont[Scale=1.10]{\cvIconFace}
  \tracinglostchars=1
\fi
