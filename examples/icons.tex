% The only place in this repository where an icon is named.  Both documents call
% the four commands below and never a glyph name, so the whole document moves to
% another icon set, or another style of it, with two switches given before the
% document is read:
%
%   \def\cvIconSet{material}       % fa (the default) or material
%   \def\cvIconStyle{outlined}     % the default; or filled, round, sharp, twotone
%
% `make cv-ar ICONS=material STYLE=outlined` passes exactly those lines on the
% XeLaTeX command line, so nothing in the sources changes to see another set or
% style.  The family each set draws with is named in fonts.tex; this file only
% picks the glyphs.
%
% Font Awesome arrives with its CTAN package, so it needs no font file, and it is
% the default set: `outlined` is its regular style, and the package substitutes the
% solid glyph wherever the requested style has no cut — a brand mark, which only
% exists in brands, stays as it was.
%
% Material Icons is Google's set and install.sh installs all five of its families.
% It draws an icon from its *name* as a ligature — type `mail` and an envelope
% appears — and a name it does not know typesets as nothing at all rather than as a
% visible box, so a typo here would be invisible in the PDF.
% tests/check-arabic-examples.sh renders every name in this file and fails on a
% blank one.
%
% Material Icons has no brand marks: `code` stands in for the GitHub profile and
% `work` for the LinkedIn one, the nearest shapes it has.
%-------------------------------------------------------------------------------
\ifx\cvIconSet\cvIconSetMaterial
  % Two adjustments the family needs, both measured against the Font Awesome row at
  % the same size: its glyphs are drawn about 0.2em above the baseline, so the
  % \raisebox drops them on to it, and the family has no space glyph, so the first
  % use of the font makes XeTeX set its interword space up and log "Missing
  % character: There is no (U+0020)".  \tracinglostchars is local, so the group
  % keeps that one line out of the log without hiding a missing character anywhere
  % else in the document.
  \newcommand*{\cviconglyph}[1]{{\tracinglostchars=0 \raisebox{-0.2em}{\iconfont #1}}}
  \newcommand*{\cvIconPhone}{\cviconglyph{smartphone}}
  \newcommand*{\cvIconMail}{\cviconglyph{mail}}
  \newcommand*{\cvIconGithub}{\cviconglyph{code}}
  \newcommand*{\cvIconLinkedin}{\cviconglyph{work}}
\else
  \usepackage{fontawesome7}
  \ifx\cvIconStyle\cvIconStyleOutlined \faStyle{regular}\else\faStyle{solid}\fi
  \newcommand*{\cvIconPhone}{\faIcon{mobile}}
  \newcommand*{\cvIconMail}{\faIcon{envelope}}
  \newcommand*{\cvIconGithub}{\faIcon{square-github}}
  \newcommand*{\cvIconLinkedin}{\faIcon{linkedin}}
\fi
